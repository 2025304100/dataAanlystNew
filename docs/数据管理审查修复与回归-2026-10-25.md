```sql

  AND b.universe_symbol_id IN (SELECT id FROM universe_symbols WHERE region='cn' AND asset_type='etf')

       SUM(s.id IS NOT NULL AND f.id IS NULL) AS no_capital_flow

JOIN universe_symbols u ON u.id=ub.universe_symbol_id AND u.region='cn' AND u.asset_type='stock'

WHERE ub.trade_date BETWEEN '2026-10-12' AND '2026-10-19'
GROUP BY ub.trade_date ORDER BY ub.trade_date;
```
