def get_latest_version(db: Session, factor_id: int) -> FactorVersion | None:
    """获取因子的当前最新版本（is_latest=1）。"""
    return db.execute(
        select(FactorVersion)
        .where(
            FactorVersion.factor_id == factor_id,
            FactorVersion.is_latest == 1,
        )
        .limit(1)
    ).scalar_one_or_none()
